<script> ::=

<library_definition> ::= 'library' <identifier> '{' <library_body> '}'
<library_body> ::= {<statement>}
<library_import> ::= 'import' '<' <scoped_type> '>' '('  ')'

<class_definition> ::= 'class' <class_name> ':' <scoped_type> <class_body>
<class_body> ::= '{' {<variable_creation> ';'} '}'
<class_name> ::= <identifier>
<class_variable> ::= <identifier> {<member_access_operator> <identifier>}

<function_definition> ::= <scoped_type>'(' <function_parameters> ')' '\s' <identifier> <function_body>
<function_body> ::= '{' {{<expression>} {<statement>}} 'return' <expression> '}'
<function_call> ::= <identifier> '(' <function_arguments> ')'
<function_parameters> ::= [<complex_type> {',' <complex_type>}]
<function_arguments> ::= [<expression> {',' <expression>}]

<statement> ::= <variable_creation> ';' | <function_definition> ';' | <class_definition> ';' | <library_definition> ';'
<expression> ::= <function_call> ';' | <variable> ';'
<variable_creation> ::= <scoped_type> '\s' <identifier>[<initializer>]
<complex_type> ::= <scoped_type> {'(' {<complex_type> {',' <complex_type>}} ')' [<reference_operator>]}
<scoped_type> ::= <type> {'::' <type>} [<reference_operator>]
<type> ::= <built_in_type> | <class_name>
<initializer> ::= '{' <variable> {',' <variable>} '}'
<variable> ::= <literal> | <identifier> | <class_variable> | <reference_operator> <identifier> | <reference_operator> <class_variable>
<literal> ::= <bool_literal> | <char_literal> | <integer_literal> | <float_literal> | <string_literal>
<template_fill> ::= '<' <complex_type> [<template_fill] '>'

<reference_operator> ::= '&'
<member_access_operator> ::= '.'
<built_in_type> ::= 'bool' | 'char' | 'int' | 'float' | 'string' | 'Object'
<identifier> ::= a-zA-Z-z\_[a-zA-Z0-9\_]
<bool_literal> ::= 'true' | 'false'
<char_literal> ::= '\'' . '\''
<integer_literal> ::= 0-9+
<float_literal> ::= 0-9+ '.' 0-9+
<string_literal> ::= '"' .* '"'
