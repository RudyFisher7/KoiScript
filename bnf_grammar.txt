<script> ::=

<library_definition> ::= 'library' <type_name> <library_body>
<library_body> ::= '{' {<statement>} '}' ';'
<library_import> ::= 'import' '<' <scoped_type> '>' '('  ')' ';'

<class_definition> ::= 'class' <class_type_name> <inheritance_operator> <template_class_type> <class_body>
    | 'class' <class_type_name> <class_template_definition> <inheritance_operator> <template_class_type_specialization> <template_class_body>
<class_body> ::= '{' {<variable_creation>} '}' ';'
<template_class_body> ::= '{' {<template_variable_creation>} '}' ';'
<class_variable_access> ::= <identifier> {<member_access_operator> <identifier>}
<template_class_type_specialization> ::= <class_type> <class_template_fill>
<class_template_definition> ::= '<' <identifier> <inheritance_operator> <template_class_type_specialization> {',' <identifier> <inheritance_operator> <template_class_type_specialization>} [','] '>'
<variable_creation> ::= 'var' <identifier> <template_class_type> [<initializer>] ';'
<template_variable_creation> ::= 'var' <identifier> <template_class_type> [<initializer>] ';'
    | 'var' <identifier> <identifier> [<initializer>] ';'

<function_definition> ::= 'func' <identifier> <function_signature> <function_body>
    | 'func' <identifier> <template_definition> <template_function_signature> <function_body>
<anonymous_function_creation> ::= <function_signature> <function_body>
    | 'func' <template_definition> <template_function_signature> <function_body>
<function_signature> ::= <scoped_type>'(' <function_parameters> ')'
<template_function_signature> ::= <scoped_type>'(' <function_parameters> ')'
    | <identifier>'(' <template_function_parameters> ')'
<function_body> ::= '{' {{<expression> ';'} {<statement> ';'}} <return_operator> <expression> ';' '}' ';'
<function_call> ::= <identifier> '(' <function_arguments> ')' ';'
<template_function_call> ::= <identifier> '(' <function_arguments> ')' ';' | <identifier> <template_fill> '(' <function_arguments> ')' ';'
<function_parameters> ::= [<any_type> <identifier> {',' <any_type> <identifier>} [',']]
<template_function_parameters> ::= [<any_type> <identifier> {',' <any_type> <identifier>} [',']]
    | [<identifier> <identifier> {<template_another_function_parameter>} [',']]
<template_another_function_parameter> ::= {',' <any_type> <identifier>} | {',' <identifier> <identifier>}
<function_arguments> ::= [<expression> {',' <expression>} [',']]

<any_type> ::= <class_type> | <function_type>
<scoped_type> ::= <type> {'::' <type>} [<reference_operator>]

<statement> ::= <variable_creation>
<expression> ::= <function_call> | <variable_access> | <anonymous_function_creation> | <anonymous_variable_creation>

<class_template_fill> ::= '<'  '>'

<reference_operator> ::= '&'
<member_access_operator> ::= '.'
<return_operator> ::= 'return'
<inheritance_operator> ::= ':'
<scope_operator> ::= '::'
<class_type> ::= 'bool_t' | 'char_t' | 'number_t' | 'int_t' | 'float_t' | 'string_t' | 'object_t' | <class_type_name>
<class_type_name> ::= a-zA-Z\_+ '_t'
<identifier> ::= a-zA-Z-z\_ [a-zA-Z0-9\_*] ^\_t
<bool_literal> ::= 'true' | 'false'
<char_literal> ::= '\'' . '\''
<integer_literal> ::= 0-9+
<float_literal> ::= 0-9+ '.' 0-9+
<string_literal> ::= '"' .* '"'
